# Code patch sets of Super Mario RPG 1.0.1 (recomp/patches.py describes the format; gen.py --patches reads this file).
# Module offsets are in main. The patch page's words in use: +32 the output width, +36 its height
# (runtime/resolution.c, res_init).

# Output size. The Unity player's default resolution routine (main+0x6b7204, called at start and when the console
# mode changes, where it logs "Operation mode changed to console, switching to: %ix%i") stores 1280x720 and then, docked
# (GetOperationMode and GetPerformanceMode both 1, or either alone depending on a player setting), 1920x1080 through
# its two int pointers. The four constant loads of the docked size read the page instead. The handheld size stays.
set output   MRPG_PATCH_OUTPUT   0
set scene    MRPG_PATCH_SCENE    0   # no sites: Unity renders its scene at the output size
site output main 6b72b8 5280f008 @page:x8 b9402108   # mov w8, #0x780 -> adrp x8, page; ldr w8, [x8, #32]
site output main 6b72c0 52808709 @page:x9 b9402529   # mov w9, #0x438 -> adrp x9, page; ldr w9, [x9, #36]
site output main 6b72d0 5280f008 @page:x8 b9402108   # mov w8, #0x780 -> adrp x8, page; ldr w8, [x8, #32]
site output main 6b72d8 52808708 @page:x8 b9402508   # mov w8, #0x438 -> adrp x8, page; ldr w8, [x8, #36]

# The player clamps a requested resolution to 1920x1080 (main+0x62f2c8: min(width, 1920) to +0xc0, min(height, 1080)
# to +0xc4 of its screen object): the limits come from the page. x9 is dead here (a call follows).
site output main 62f2c8 711e02bf @page:x8 b9402108 6b0802bf   # cmp w21, #0x780 -> w8 = [page+32]; cmp w21, w8
site output main 62f2cc 5280f008 2a0803e8                     # mov w8, #0x780 -> mov w8, w8
site output main 62f2d4 7110e29f @page:x9 b9402529 6b09029f   # cmp w20, #0x438 -> w9 = [page+36]; cmp w20, w9
site output main 62f2dc 52808708 2a0903e8                     # mov w8, #0x438 -> mov w8, w9

# The 1920x1080 images the player makes whatever the resolution: a render surface through the device's virtual call
# at main+0x6b60e0, and two NVN texture builders' sizes (main+0x6caf28 and +0x6cb1a0; the window's three images are
# among them: nvnWindowInitialize logged 1920x1080 with only the routine above patched, and the frame was scaled down
# into them).
site output main 6b60bc 52808703 @page:x3 b9402463    # mov w3, #0x438 -> ldr w3, [page+36]
site output main 6b60c8 5280f002 @page:x2 b9402042    # mov w2, #0x780 -> ldr w2, [page+32]
site output main 6caf18 52808717 @page:x23 b94026f7   # mov w23, #0x438 -> ldr w23, [page+36]
site output main 6caf20 5280f001 @page:x1 b9402021    # mov w1, #0x780 -> ldr w1, [page+32]
site output main 6cb198 5280f001 @page:x1 b9402021    # mov w1, #0x780 -> ldr w1, [page+32]
site output main 6cb19c 52808702 @page:x2 b9402442    # mov w2, #0x438 -> ldr w2, [page+36]

# A shader parameter set to (1920, 1080) as two floats (main+0xb7c064): the pair comes from page +32 as one word.
site output main b7c064 d280f001 @page:x1             # mov x1, #0x780 -> adrp x1, page
site output main b7c068 f2c08701 f9401021             # movk x1, #0x438, lsl #32 -> ldr x1, [x1, #32]
