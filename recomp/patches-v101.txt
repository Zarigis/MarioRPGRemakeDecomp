# Code patch sets of Super Mario RPG 1.0.1 (recomp/patches.py describes the format; gen.py --patches reads this file).
# Module offsets are in main. The patch page's words in use: +32 the output width, +36 its height
# (runtime/resolution.c, res_init); +64..83 the walking speed by stick tilt and the movement dead zone (runtime/input.c, move_tilt); +84 the shadow map scale
# (runtime/resolution.c, res_init); +88..99 the camera offsets (runtime/input.c, camera_stick).

# Output size. The Unity player's default resolution routine (main+0x6b7204, called at start and when the console
# mode changes, where it logs "Operation mode changed to console, switching to: %ix%i") stores 1280x720 and then, docked
# (GetOperationMode and GetPerformanceMode both 1, or either alone depending on a player setting), 1920x1080 through
# its two int pointers. The four constant loads of the docked size read the page instead. The handheld size stays.
set output   MRPG_PATCH_OUTPUT   0
set scene    MRPG_PATCH_SCENE    0   # no sites: Unity renders its scene at the output size
site output main 6b72b8 5280f008 @page:x8 b9402108   # mov w8, #0x780 -> adrp x8, page; ldr w8, [x8, #32]
site output main 6b72c0 52808709 @page:x9 b9402529   # mov w9, #0x438 -> adrp x9, page; ldr w9, [x9, #36]
site output main 6b72d0 5280f008 @page:x8 b9402108   # mov w8, #0x780 -> adrp x8, page; ldr w8, [x8, #32]
site output main 6b72d8 52808708 @page:x8 b9402508   # mov w8, #0x438 -> adrp x8, page; ldr w8, [x8, #36]

# The player clamps a requested resolution to 1920x1080 (main+0x62f2c8: min(width, 1920) to +0xc0, min(height, 1080)
# to +0xc4 of its screen object): the limits come from the page. x9 is dead here (a call follows).
site output main 62f2c8 711e02bf @page:x8 b9402108 6b0802bf   # cmp w21, #0x780 -> w8 = [page+32]; cmp w21, w8
site output main 62f2cc 5280f008 2a0803e8                     # mov w8, #0x780 -> mov w8, w8
site output main 62f2d4 7110e29f @page:x9 b9402529 6b09029f   # cmp w20, #0x438 -> w9 = [page+36]; cmp w20, w9
site output main 62f2dc 52808708 2a0903e8                     # mov w8, #0x438 -> mov w8, w9

# The 1920x1080 images the player makes whatever the resolution: a render surface through the device's virtual call
# at main+0x6b60e0, and two NVN texture builders' sizes (main+0x6caf28 and +0x6cb1a0; the window's three images are
# among them: nvnWindowInitialize logged 1920x1080 with only the routine above patched, and the frame was scaled down
# into them).
site output main 6b60bc 52808703 @page:x3 b9402463    # mov w3, #0x438 -> ldr w3, [page+36]
site output main 6b60c8 5280f002 @page:x2 b9402042    # mov w2, #0x780 -> ldr w2, [page+32]
site output main 6caf18 52808717 @page:x23 b94026f7   # mov w23, #0x438 -> ldr w23, [page+36]
site output main 6caf20 5280f001 @page:x1 b9402021    # mov w1, #0x780 -> ldr w1, [page+32]
site output main 6cb198 5280f001 @page:x1 b9402021    # mov w1, #0x780 -> ldr w1, [page+32]
site output main 6cb19c 52808702 @page:x2 b9402442    # mov w2, #0x438 -> ldr w2, [page+36]

# A shader parameter set to (1920, 1080) as two floats (main+0xb7c064): the pair comes from page +32 as one word.
site output main b7c064 d280f001 @page:x1             # mov x1, #0x780 -> adrp x1, page
site output main b7c068 f2c08701 f9401021             # movk x1, #0x438, lsl #32 -> ldr x1, [x1, #32]

# Walking speed by stick tilt (runtime/input.c, move_tilt). FieldCharacter.UpdatePlayerDirectionMove (main+0x2195d50)
# picks the player's speed: the walk speed from the field speed table (+0x30 of the class statics reached through
# main+0x3a5f070), replaced by the run speed (+0x34) when Button.get_Run (main+0x1b2ad70) says so: with "auto dash"
# on that is true unless L, R, ZL or ZR is held. The stick's tilt only sets the direction (the game normalises it, and
# ignores a tilt under 0.4). The three sites let the runtime decide instead: the get_Run result is masked and set from
# the page (+0x44 mask, +0x48 bits: and, then or; mask all ones and bits 0 leave the game's choice), and the walk and
# run speeds are scaled by the floats at +0x40 and +0x4c. x8, x9, x10 and s0 are free at these points (each is written
# before it is read on every path after them). The page's defaults (factor 1, the game's choice) come from move_tilt_init.
set tilt     MRPG_STICK_SPEED    1
site tilt main 2195e3c 36000320 @page:x8 2948a909 0a090000 2a0a0000 36000320   # tbz w0 (get_Run) -> w0 = w0 & [page+0x44] | [page+0x48]; tbz
site tilt main 2195e5c bd40350b bd40350b @page:x8 bd404d00 1e20096b            # ldr s11, run speed -> the same; s11 *= [page+0x4c]
site tilt main 2195ea0 528000b4 528000b4 @page:x8 bd404100 1e20096b            # mov w20, #5 (walk) -> the same; s11 *= [page+0x40]
# The movement dead zone: the player stands while the stick's direction vector has a squared length under 0.16 (a tilt
# of 0.4), a float the two words below build in w8 for the compare at main+0x2195f94. The bits come from page +0x50
# instead (MRPG_MOVE_DEADZONE, the tilt in percent, squared).
site tilt main 2195f7c 529ae148 @page:x8                                        # mov w8, #0xd70a -> adrp x8, page
site tilt main 2195f80 72a7c468 b9405108                                        # movk w8, #0x3e23, lsl #16 -> ldr w8, [x8, #0x50]

# Free movement direction. After turning the stick's direction by the camera, FieldCharacter.UpdatePlayerDirectionMove
# (main+0x2196044, the same test as FieldCharacter.FixInputAxis, inlined) drops the smaller of the two ground axes when
# it is under 0.6 of the larger: a direction within 31 degrees of either world axis goes along that axis. The two
# selects that zero an axis keep it instead.
set freedir  MRPG_FREE_DIRECTION 1
site freedir main 2196060 1e284d68 1e284d08   # fcsel s8, s11 (0), s8, mi -> fcsel s8, s8, s8, mi
site freedir main 219606c 1e294d69 1e294d29   # fcsel s9, s11 (0), s9, mi -> fcsel s9, s9, s9, mi

# Shadow map size. UniversalRenderPipeline.InitializeShadowData (main+0x19b9000) copies the URP asset's main light
# shadow map resolution (+0x7c of the asset: 2048 here, a 2048x1024 atlas of two cascades in battles) into the frame's
# shadow data as its width and height. Both reads multiply it by the integer at page +0x54 (MRPG_SHADOW_MAP_SCALE,
# runtime/resolution.c res_init; the page must hold 1 at least before the first frame). x10 is not read in the rest of
# the function.
set shadowmap MRPG_PATCH_SHADOWMAP 1
site shadowmap main 19b95c4 b9407e89 b9407e89 @page:x10 b940554a 1b0a7d29   # ldr w9, [x20, #0x7c] -> the same; w9 *= [page+0x54]
site shadowmap main 19b95cc b9407e89 b9407e89 @page:x10 b940554a 1b0a7d29   # the same for the height

# Camera control. FieldCamera.UpdateCommon (main+0x1d61fd0) takes the field camera's angles (x pitch, y yaw, z roll, in
# degrees) and distance from UpdateAngleDistance in s0..s3, turns the angles to radians and places the camera at that
# distance from its target. Before that, the pitch and yaw get the offsets at page +0x58 and +0x5c added and the
# distance is multiplied by +0x60 (runtime/input.c camera_stick: the right stick; 0, 0 and 1 leave the game's camera).
# The player's movement turns with the camera's yaw on its own (UpdatePlayerDirectionMove reads the camera transform).
# s4..s6 are free here (the next call clobbers them); x8 is written again by the original word, last.
set camera   MRPG_CAMERA_STICK   1
site camera main 1d62024 529f46a8 @page:x8 2d4b1504 bd406106 1e242800 1e252821 1e260863 529f46a8   # w8 = 0x...fa35 -> offsets first
